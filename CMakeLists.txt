cmake_minimum_required(VERSION 3.16)
project(PUTM_EV_CAN_LIBRARY LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)

# --- Backend Detection ---
if(NOT DEFINED PUTM_CAN_BACKEND)
    if(CMAKE_SYSTEM_NAME STREQUAL "Generic")
        set(PUTM_CAN_BACKEND STM32)
    elseif(DEFINED ENV{AMENT_PREFIX_PATH})
        set(PUTM_CAN_BACKEND ROS2)
    else()
        set(PUTM_CAN_BACKEND STM32)
    endif()
endif()

message(STATUS "Selected Backend: ${PUTM_CAN_BACKEND}")

# --- DBC Code Generation Setup ---
find_package(Python3 REQUIRED COMPONENTS Interpreter)

set(CAN_GEN_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/database/setup.py")
set(CAN_GEN_WORKDIR "${CMAKE_CURRENT_SOURCE_DIR}/database")
set(GEN_OUTPUT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/database/generated")

# List of expected files to be generated by the Python script.
# This tells Ninja/Make exactly what to look for after the script runs.
set(GENERATED_DBC_SOURCES
    "${GEN_OUTPUT_DIR}/PUTM_CAN_M.c"
    "${GEN_OUTPUT_DIR}/PUTM_CAN_PT.c"
    #"${GEN_OUTPUT_DIR}/PUTM_CAN_DV.c" !!!!!!!!!!!!uncomment if there will be a dv file
)

# Define the command to generate the files. 
# The build system will only run this if the OUTPUT files are missing or the DEPENDS script changes.
add_custom_command(
    OUTPUT ${GENERATED_DBC_SOURCES}
    COMMAND ${Python3_EXECUTABLE} "${CAN_GEN_SCRIPT}"
    WORKING_DIRECTORY "${CAN_GEN_WORKDIR}"
    DEPENDS "${CAN_GEN_SCRIPT}"
    COMMENT "Running CAN Database Generator (setup.py)..."
    VERBATIM
)

# This target acts as a build-time trigger for the command above
add_custom_target(
    GenerateCanCode ALL
    DEPENDS ${GENERATED_DBC_SOURCES}
)

# --- Source Selection ---
# We mark these as GENERATED so CMake doesn't check for their existence during configuration
set_source_files_properties(${GENERATED_DBC_SOURCES} PROPERTIES GENERATED TRUE)

set(COMMON_SOURCES
    "${CMAKE_CURRENT_SOURCE_DIR}/src/can_driver.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/message_handler.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/dbc_registry.cpp"
    ${GENERATED_DBC_SOURCES}
)

set(HAL_SOURCES "")
if(PUTM_CAN_BACKEND STREQUAL "STM32")
    list(APPEND HAL_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/can_hal_stm32.cpp")
elseif(PUTM_CAN_BACKEND STREQUAL "ROS2")
    list(APPEND HAL_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/can_hal_ros2.cpp")
endif()

# --- Library Definition ---
add_library(putm_ev_can STATIC ${COMMON_SOURCES} ${HAL_SOURCES})

# Explicit dependency ensures GenerateCanCode finishes before putm_ev_can starts compiling
add_dependencies(putm_ev_can GenerateCanCode)

# --- Includes & Dependencies ---
target_include_directories(putm_ev_can PUBLIC 
    include 
    database/generated
    ${CMAKE_CURRENT_SOURCE_DIR}
)

if(PUTM_CAN_BACKEND STREQUAL "STM32")
    target_compile_definitions(putm_ev_can PUBLIC PUTM_CAN_BACKEND_STM32)
    
    if(TARGET stm32cubemx)
        target_link_libraries(putm_ev_can PUBLIC stm32cubemx)
    endif()

    # Generalized path detection for STM32 (G4, F4, H7, etc.)
    get_filename_component(PROJECT_ROOT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
    
    file(GLOB HAL_INC_DIR "${PROJECT_ROOT_DIR}/Drivers/STM32*xx_HAL_Driver/Inc")
    file(GLOB CMSIS_DEVICE_DIR "${PROJECT_ROOT_DIR}/Drivers/CMSIS/Device/ST/STM32*xx/Include")
    
    target_include_directories(putm_ev_can PUBLIC
        ${HAL_INC_DIR}
        ${CMSIS_DEVICE_DIR}
        "${PROJECT_ROOT_DIR}/Drivers/CMSIS/Include"
    )

elseif(PUTM_CAN_BACKEND STREQUAL "ROS2")
    target_compile_definitions(putm_ev_can PUBLIC PUTM_CAN_BACKEND_ROS2)
    find_package(Threads REQUIRED)
    find_package(rclcpp REQUIRED)
    target_link_libraries(putm_ev_can PRIVATE Threads::Threads rclcpp::rclcpp)
endif()
